<!-- defaultExercise computation (D-15):
     Count occurrences of each exercise name across all gymRuns[].exercises[],
     pick the name with the highest count. If tie, pick first alphabetically.
     Pass result as defaultExercise prop to WeightProgressionChart. -->
```typescript
function computeDefaultExercise(gymRuns: Run[]): string | undefined {
  const counts = new Map<string, number>()
  for (const run of gymRuns) {
    for (const ex of run.exercises ?? []) {
      counts.set(ex.name, (counts.get(ex.name) ?? 0) + 1)
    }
  }
  if (counts.size === 0) return undefined
  return Array.from(counts.entries())
    .sort(([a, ca], [b, cb]) => cb - ca || a.localeCompare(b))[0][0]
}
```
